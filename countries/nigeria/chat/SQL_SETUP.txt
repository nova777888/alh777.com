-- Nova Chat 数据库表
-- 在 Supabase SQL Editor 中执行

-- 1. 消息表
CREATE TABLE IF NOT EXISTS nova_messages (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  sender VARCHAR(50) NOT NULL,
  receiver VARCHAR(50) NOT NULL,
  content TEXT,
  media_url TEXT,
  media_type VARCHAR(20) CHECK (media_type IN ('image', 'video')),
  created_at TIMESTAMPTZ DEFAULT now(),
  is_read BOOLEAN DEFAULT false
);

-- 2. 索引
CREATE INDEX IF NOT EXISTS idx_nova_messages_chat ON nova_messages(sender, receiver, created_at DESC);
CREATE INDEX IF NOT EXISTS idx_nova_messages_receiver ON nova_messages(receiver, created_at DESC);

-- 3. 会话表（用于聊天列表）
CREATE TABLE IF NOT EXISTS nova_conversations (
  user_a VARCHAR(50),
  user_b VARCHAR(50),
  last_message TEXT,
  last_message_time TIMESTAMPTZ,
  unread_count INTEGER DEFAULT 0,
  PRIMARY KEY (user_a, user_b)
);

-- 4. 启用 RLS
ALTER TABLE nova_messages ENABLE ROW LEVEL SECURITY;
ALTER TABLE nova_conversations ENABLE ROW LEVEL SECURITY;

-- 5. 创建访问策略
CREATE POLICY "Public read messages" ON nova_messages FOR SELECT USING (true);
CREATE POLICY "Public insert messages" ON nova_messages FOR INSERT WITH CHECK (true);
CREATE POLICY "Public read conversations" ON nova_conversations FOR SELECT USING (true);
CREATE POLICY "Public insert conversations" ON nova_conversations FOR INSERT WITH CHECK (true);
CREATE POLICY "Public update conversations" ON nova_conversations FOR UPDATE USING (true);

-- 6. 启用 Realtime
ALTER PUBLICATION supabase_realtime ADD TABLE nova_messages;
ALTER PUBLICATION supabase_realtime ADD TABLE nova_conversations;
